create table public.endangered_wildlife (
  id uuid not null default gen_random_uuid (),
  species_name character varying(255) not null,
  description text null,
  conservation_status character varying(100) null,
  image_url text null,
  image_urls text[] null default array[]::text[],
  municipality_id uuid null,
  approval_status character varying(50) null default 'pending'::character varying,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint endangered_wildlife_pkey primary key (id),
  constraint endangered_wildlife_municipality_id_fkey foreign KEY (municipality_id) references profiles (id) on delete CASCADE,
  constraint status_check check (
    (
      (approval_status)::text = any (
        (
          array[
            'pending'::character varying,
            'approved'::character varying,
            'declined'::character varying
          ]
        )::text[]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_endangered_wildlife_municipality_id on public.endangered_wildlife using btree (municipality_id) TABLESPACE pg_default;

create index IF not exists idx_endangered_wildlife_approval_status on public.endangered_wildlife using btree (approval_status) TABLESPACE pg_default;

create index IF not exists idx_endangered_wildlife_created_at on public.endangered_wildlife using btree (created_at desc) TABLESPACE pg_default;
